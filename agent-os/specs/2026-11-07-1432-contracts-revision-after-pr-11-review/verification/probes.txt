class LoadsUntilARefusal(FakeInstrument):
    """Checks the name, then checks and stores each value in turn, publishing PatchLoaded at the end."""

class Refused(Exception):
    pass


@contextmanager
def _refusals() -> Iterator[None]:
    try:
        yield
    except ValidationError as error:
        raise Refused([".".join(map(str, e["loc"])) for e in error.errors()]) from error


def _refusing[**P, R](validate: Callable[P, R]) -> Callable[P, R]:
    @functools.wraps(validate)
    def refusing(*args: P.args, **kwargs: P.kwargs) -> R:
        with _refusals():
            return validate(*args, **kwargs)

    return refusing

import functools
from collections.abc import Callable, Iterator
from contextlib import contextmanager
from typing import TYPE_CHECKING, assert_type

from pydantic import BaseModel, ConfigDict, ValidationError


@contextmanager
def _refusals() -> Iterator[None]:
    try:
        yield
    except ValidationError as error:
        raise ValueError from error


def _refusing[**P, R](validate: Callable[P, R]) -> Callable[P, R]:
    @functools.wraps(validate)
    def refusing(*args: P.args, **kwargs: P.kwargs) -> R:
        with _refusals():
            return validate(*args, **kwargs)

    return refusing


class Base(BaseModel, frozen=True):
