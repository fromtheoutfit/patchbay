import pytest

from patchbay.models import Parameter, Quantity, Unit
from patchbay.testing import contract_tests as contract


def old_out_of_range(parameter: Parameter) -> list[Quantity]:
    """_out_of_range at 6f4e7a8, verbatim."""
    candidates = [parameter.maximum + 1.0, parameter.minimum - 1.0]
    if parameter.unit is Unit.ratio:
        candidates = [
            value for value in (1.0, 0.0) if not parameter.minimum <= value <= parameter.maximum
        ]
    if parameter.kind != "continuous":
        candidates.append(parameter.minimum + 0.5)
    return [Quantity(value=value, unit=parameter.unit) for value in candidates]


contract._out_of_range = old_out_of_range  # the mutation: finding 6's code back in place
