A set goes display -> `MappedParameter.to_normalized` (the curve inverted and rounded to a float32
preset, then that rounded to a float32 normalized value) -> the plugin's `fma32` back to a preset;
the display then held is `MappedParameter.from_normalized` of the normalized value. Each case reads
the committed map, `parameter_map()`, and writes nothing.

import numpy as np

m = parameter_map()


def neighbours(r: np.float32) -> list[float]:
    """The float32 normalized values either side of `r`, and `r`, inside [0, 1]."""
    side = (np.nextafter(r, np.float32(-1)), r, np.nextafter(r, np.float32(2)))
    return [float(n) for n in side if 0.0 <= float(n) <= 1.0]

C1 (finding 18): probe a's "a value get returned" check, given a positive control. The recorder is

`MovesInterior` holds bounds exactly but moves an interior whole number it is set to; the old

them apart.
"""

import math

from patchbay.errors import PatchbayError
from patchbay.events import EventBus
from patchbay.models import Parameter, ParameterId, Quantity, Unit
from patchbay.testing import CONTRACT_TESTS, FakeInstrument
from patchbay.testing.fake import PARAMETERS

# --- C1 -------------------------------------------------------------------------------------------


class Recording(FakeInstrument):
    """Probe a's recorder: each set, and whether its value is one `get` returned in the test."""

    def __init__(self, bus: EventBus) -> None:
        self.log: list[str] = []
        self.got: dict[ParameterId, list[float]] = {}
        super().__init__(bus)

    def get(self, parameter_id: ParameterId, /) -> Quantity:
        value = super().get(parameter_id)
        self.got.setdefault(parameter_id, []).append(value.value)
        return value

    def set(self, parameter_id: ParameterId, value: Quantity, /) -> None:
        returned = value.value in self.got.get(parameter_id, [])
        try:
            super().set(parameter_id, value)
        except PatchbayError:
            self.log.append(f"set {parameter_id} {value.value!r}: refused")
            raise
        note = ", a value get returned" if returned else ""
        self.log.append(f"set {parameter_id} {value.value!r}: accepted{note}")


print("== C1: probe a's check, with a positive control (finding 18)")
control = Recording(EventBus())
level = ParameterId("level")
control.set(level, control.get(level))
print(f"control, set(level, get(level)): {control.log[-1]}")
print(f"  the check fires: {'a value get returned' in control.log[-1]}")
fired = 0
for test in CONTRACT_TESTS:
    bus = EventBus()
    instrument = Recording(bus)
    test(instrument, bus)
    fired += sum("a value get returned" in line for line in instrument.log)
print(f"the thirteen contract tests: the check fires {fired} times")

VOICES = ParameterId("voices")
VOICES_PARAMETER = Parameter(
    id=VOICES,
    label="Voices",
    kind="continuous",
    unit=Unit.none,
    minimum=1.0,
    maximum=16.0,
    default=Quantity(value=1.0, unit=Unit.none),
)


def half_away(value: float) -> float:
    return math.copysign(math.floor(abs(value) + 0.5), value)


class Honest(FakeInstrument):
    """Probe b's fake: holds whole numbers, rounded a half away from zero."""

    def parameters(self) -> tuple[Parameter, ...]:
        return (*PARAMETERS, VOICES_PARAMETER)

    def hold(self, value: float) -> float:
        return half_away(value)

    def set(self, parameter_id: ParameterId, value: Quantity, /) -> None:
        if parameter_id == VOICES:
            VOICES_PARAMETER.check(value)
            value = Quantity(value=self.hold(value.value), unit=value.unit)
        super().set(parameter_id, value)


class MovesInterior(Honest):
    """Holds bounds exactly, but an interior whole number it is set to moves up one."""

    def hold(self, value: float) -> float:
        if value.is_integer() and VOICES_PARAMETER.minimum < value < VOICES_PARAMETER.maximum:
            return value + 1.0
        return half_away(value)


def old_sequence(instrument: Honest) -> list[str]:
    lines = []
    for given in (16.0, 1.0, 3.4, 3.5, 2.5, 3.6, 15.5):
        instrument.set(VOICES, Quantity(value=given, unit=Unit.none))
        lines.append(f"set {given!r:>5} -> get {instrument.get(VOICES).value!r:>5}")
    again = instrument.get(VOICES)
    instrument.set(VOICES, again)
    lines.append(f"set(get()) {again.value!r} -> get {instrument.get(VOICES).value!r}")
    return lines


def after_interior(instrument: Honest) -> str:
    instrument.set(VOICES, Quantity(value=3.4, unit=Unit.none))
    again = instrument.get(VOICES)
    instrument.set(VOICES, again)
    return f"set 3.4, then set(get()) {again.value!r} -> get {instrument.get(VOICES).value!r}"
