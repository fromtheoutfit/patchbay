== probe_a_values_given.py

"""Probe a: every value each contract test gives FakeInstrument, accepted or refused.

For every `set` the probe records the parameter, the value, whether the instrument took it, whether
it is one of the parameter's bounds, and whether it equals a value `get` returned for that parameter
earlier in the same test. For every `load_patch` it records whether the patch loaded and, for one
that did, whether each value in it is one `patch()` had returned (a value the instrument holds).
"""

from patchbay.errors import PatchbayError
from patchbay.events import EventBus
from patchbay.models import ParameterId, Patch, Quantity
from patchbay.testing import CONTRACT_TESTS, FakeInstrument


class Recording(FakeInstrument):
    def __init__(self, bus: EventBus) -> None:
        self.log: list[str] = []
        self.got: dict[ParameterId, list[float]] = {}
        self.patched: list[Patch] = []
        super().__init__(bus)

    def get(self, parameter_id: ParameterId, /) -> Quantity:
        value = super().get(parameter_id)
        self.got.setdefault(parameter_id, []).append(value.value)
        return value

    def set(self, parameter_id: ParameterId, value: Quantity, /) -> None:
        parameter = next((p for p in self.parameters() if p.id == parameter_id), None)
        if parameter is None:
            bound = "no such id"
        elif value.unit is not parameter.unit:
            bound = f"unit {value.unit.value}, not {parameter.unit.value}"
        elif value.value in (parameter.minimum, parameter.maximum):
            bound = "at a bound"
        else:
            bound = f"NOT a bound ({parameter.kind}, [{parameter.minimum}, {parameter.maximum}])"
        returned = value.value in self.got.get(parameter_id, [])
        try:
            super().set(parameter_id, value)
        except PatchbayError as error:
            self.log.append(f"  set {parameter_id} {value.value!r}: refused ({type(error).__name__}), {bound}")
            raise
        note = ", a value get returned" if returned else ""
        self.log.append(f"  set {parameter_id} {value.value!r}: accepted, {bound}{note}")

    def patch(self) -> Patch:
        patch = super().patch()
        self.patched.append(patch)
        return patch

    def load_patch(self, patch: Patch, /) -> None:
        held = any(patch.values == earlier.values for earlier in self.patched)
        try:
            super().load_patch(patch)
        except PatchbayError as error:
            self.log.append(f"  load_patch of {len(patch.values)} values: refused ({type(error).__name__})")
            raise
        source = "every value one patch() returned" if held else "values patch() never returned"
        self.log.append(f"  load_patch of {len(patch.values)} values: accepted, {source}")


accepted_off_bound = 0
for test in CONTRACT_TESTS:
    bus = EventBus()
    instrument = Recording(bus)
    test(instrument, bus)
    print(f"{test.__name__}:")
    print("\n".join(instrument.log) if instrument.log else "  no set, no load_patch")
    accepted_off_bound += sum(
        1 for line in instrument.log if "set " in line and "accepted" in line and "NOT a bound" in line
    )
print()
print(f"accepted sets not at a bound: {accepted_off_bound}")

== probe_b_whole_numbers.py

"""Probe b: a parameter that takes whole numbers on a continuous range, against the wording.

`Whole` is FakeInstrument with one more parameter, `voices`: continuous, unit none, 1 to 16. Like
patchbay-vital's indexed scales it holds whole numbers only: `set` checks the value against the
parameter first, then holds it rounded a half away from zero (`std::round`), and `load_patch` holds
each `voices` value so too after the whole patch is checked. Every contract test runs against it;
then the values the wording speaks of are set and read back.
"""

import math

from patchbay.events import EventBus, ParameterChanged
from patchbay.models import Parameter, ParameterId, Patch, Quantity, Unit
from patchbay.testing import CONTRACT_TESTS, FakeInstrument, RecordingSubscriber
from patchbay.testing.fake import PARAMETERS

VOICES = ParameterId("voices")


def half_away(value: float) -> float:
    return math.copysign(math.floor(abs(value) + 0.5), value)


class Whole(FakeInstrument):
    def parameters(self) -> tuple[Parameter, ...]:
        voices = Parameter(
            id=VOICES,
            label="Voices",
            kind="continuous",
            unit=Unit.none,
            minimum=1.0,
            maximum=16.0,
            default=Quantity(value=1.0, unit=Unit.none),
        )
        return (*PARAMETERS, voices)

    def set(self, parameter_id: ParameterId, value: Quantity, /) -> None:
        if parameter_id == VOICES:
            self._parameter(parameter_id).check(value)
            value = Quantity(value=half_away(value.value), unit=value.unit)
        super().set(parameter_id, value)

    def load_patch(self, patch: Patch, /) -> None:
        self._check_whole(patch)
        values = dict(patch.values)
        if VOICES in values:
            values[VOICES] = Quantity(value=half_away(values[VOICES].value), unit=Unit.none)
        super().load_patch(Patch(instrument=patch.instrument, values=values, opaque=patch.opaque))


for test in CONTRACT_TESTS:
    bus = EventBus()
    test(Whole(bus), bus)
    print(f"passes  {test.__name__}")
